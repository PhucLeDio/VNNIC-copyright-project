rectangle "Target URL / Domain\n(Đầu vào cần quét)" as input <<input>>

package "STEP 0 (Part 1): Whitelist Check" {
    rectangle "VNNIC Whitelist CSV\n(Kiểm tra CSDL tên miền hợp pháp)" as db_whitelist
}

package "STEP 1: Perimeter Guard & Network Scan" {
    rectangle "WHOIS Lookup\n(Tuổi domain, Registrar, WHOIS Privacy)" as whois_scan
    rectangle "DNS Query Engine\n(Bản ghi A, AAAA, MX, NS, TXT)" as dns_query
    rectangle "Redirect Chain Tracking\n(Bắt chuỗi chuyển hướng & Cross-domain)" as redirect_tracker
    rectangle "IP & ASN Resolver\n(Phát hiện Cloud Provider & Suspicious Hosting)" as ip_asn_resolver
    rectangle "Network Risk & Legitimacy Calculator\n(Tính điểm rủi ro TLD, Mail, Legitimate Signals)" as risk_calculator
}

package "STEP 0 (Part 2): Domain Model Pre-screening" {
    rectangle "Feature Extractor\n(Trích xuất 12 đặc trưng Lexical + Hosting từ Step 1)" as feat_extractor
    rectangle "HybridFeaturesDomain Model\n(PhoBERT + 12 Features Classification)" as model_domain <<model>>
    rectangle "Domain Pre-screening Flag\n(FLAG_DOMAIN_PREDICT_MALICIOUS / OK)" as domain_flag
}

package "STEP 2: Deep Browser Evidence Collection (Playwright)" {
    rectangle "Playwright-Stealth Browser\n(Vượt rào Cloudflare JS challenge)" as browser_crawl
    rectangle "Deep Watch/Episode Pages Scan\n(Trích xuất URL tập phim/nguồn phát)" as watch_pages
    rectangle "Network Stream Interceptor\n(Bắt luồng .m3u8, .mp4, .ts, .mpd & DRM)" as stream_interceptor
    rectangle "Embedded iFrames Resolver\n(Giải mã iFrames phát video ẩn)" as iframe_resolver
    rectangle "Ad Banner Collector & Network Hits\n(Thu thập banner DOM & URL quảng cáo)" as banner_collector
    rectangle "DOM Text Extractor\n(Trích xuất văn bản thô từ DOM cho NLP)" as dom_extractor
    rectangle "Legal Footer & Identity Scan\n(Trích xuất MST, Giấy phép, Telegram, Ad-contact)" as footer_scan
    rectangle "Anti-Popup Block & Fullpage Screenshot\n(Tự động chặn tab quảng cáo, chụp màn hình)" as screenshots
}

package "STEP 3: AI Evidence Engines" {
    package "Branch 1: OCR Banner Pipeline (Doanh thu cờ bạc)" {
        rectangle "YOLO Banner Pre-filter\n(Model banner_detection.pt: betting vs non-betting)" as yolo_filter <<model>>
        rectangle "EasyOCR Text Extractor\n(Nhận dạng chữ tiếng Việt + Anh, hỗ trợ GIF frame)" as ocr_extractor
        rectangle "Text Normalizer\n(Lowercase, NFD loại bỏ dấu tiếng Việt)" as ocr_normalizer
        
        package "3-Tier Cascade Matcher (Fail-Fast)" {
            rectangle "Tier 1: Exact Keyword Match\n(Keywords_v1.csv)" as ocr_t1
            rectangle "Tier 2: Fuzzy String Match\n(RapidFuzz partial_ratio >= 75)" as ocr_t2
            rectangle "Tier 3: Semantic Embedding Match\n(paraphrase-multilingual-MiniLM-L12-v2 >= 0.75)" as ocr_t3 <<model>>
        }
    }
    
    package "Branch 2: Content Model (NLP DOM Text)" {
        rectangle "DOM Text Segmenter\n(Cắt 256 từ + Overlap ngắt câu)" as dom_segmenter
        rectangle "PhoBERT Multi-Task Core\n(Model content_model fine-tuned)" as phobert_core <<model>>
        rectangle "Task 1: Binary Classification\n(Safe vs. Malicious)" as task_binary
        rectangle "Task 2: Multi-Category Classification\n(10 loại: Cờ bạc, 18+, Game, Báo chí...)" as task_multiclass
        rectangle "Max Pooling & Voting Aggregator\n(Early stop độc hại & Bầu cử thể loại)" as content_aggregator
    }
}

rectangle "Evidence Buffer\n(Bể tích lũy bằng chứng tổng hợp)" as evidence_buffer <<buffer>>

package "STEP 4: Decision & Reporting Engine" {
    rectangle "Payload Reducer\n(Trích xuất ~25 trường thiết yếu, nén 60-90% token)" as payload_reducer
    rectangle "Gemini 3.5 Flash Synthesis\n(Phân tích 4 lớp vi phạm VNNIC)" as gemini_core <<model>>
    rectangle "Whitelist & Synthesis Override Logic\n(Phán quyết 3 mức: VI_PHAM / NGHI_NGO / AN_TOAN)" as override_logic
    rectangle "Report Enricher & Saver\n(Tự động tổng hợp hồ sơ số JSON vào logs & report/)" as report_enricher
}

rectangle "Synthesis Report\n(Hồ sơ bằng chứng & Phán quyết vi phạm JSON)" as final_report <<output>>
rectangle "Streamlit Monitoring Dashboard\n(Giao diện app.py trực quan hóa bằng chứng)" as dashboard <<dashboard>>
